    71→const ADMIN_PASSWORD_KEY = "raya_admin_password";
    72→
    73→function getPassword(): string {
    74→  if (typeof window === "undefined") return "";
    75→  return sessionStorage.getItem(ADMIN_PASSWORD_KEY) || "";
    76→}
    77→
    78→function setPassword(pw: string) {
    79→  if (typeof window === "undefined") return;
    80→  sessionStorage.setItem(ADMIN_PASSWORD_KEY, pw);
    81→}
    82→
    83→function clearPassword() {
    84→  if (typeof window === "undefined") return;
    85→  sessionStorage.removeItem(ADMIN_PASSWORD_KEY);
    86→}

    88→async function api(path: string, opts: RequestInit = {}) {
    89→  const pw = getPassword();
    90→  const headers: Record<string, string> = {
    91→    "content-type": "application/json",
    92→    ...(opts.headers as Record<string, string>),
    93→  };
    94→  if (pw) headers["x-admin-password"] = pw;
    95→  const res = await fetch(path, { ...opts, headers });
    96→  if (!res.ok) {
    97→    const err = await res.json().catch(() => ({}));
    98→    throw new Error(err.error || `HTTP ${res.status}`);
    99→  }
   100→  return res.json();
   101→}
